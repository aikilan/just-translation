import { chromium } from '/Users/jintao.xu/.npm/_npx/9833c18b2d85bc59/node_modules/playwright/index.mjs';
import { mkdir, writeFile } from 'node:fs/promises';
import assert from 'node:assert/strict';
const root = '/Users/jintao.xu/Documents/只是翻译';
const origin = process.env.JT_FIXTURE_ORIGIN ?? 'http://127.0.0.1:56779';
const output = root + '/artifacts/i18n';
await mkdir(output, { recursive: true });
const context = await chromium.launchPersistentContext('/tmp/jt-i18n-browser-' + Date.now(), {
  headless: true,
  executablePath: '/Users/jintao.xu/Library/Caches/ms-playwright/chromium-1243/chrome-mac-arm64/Google Chrome for Testing.app/Contents/MacOS/Google Chrome for Testing',
  args: [`--disable-extensions-except=${root}/dist`, `--load-extension=${root}/dist`],
  viewport: { width: 1280, height: 900 },
});
const errors = [];
context.on('page', page => page.on('pageerror', error => errors.push(error.message)));
const report = { locales: [], errors, checks: [] };
try {
  const worker = context.serviceWorkers()[0] ?? await context.waitForEvent('serviceworker');
  const extensionId = new URL(worker.url()).host;
  const options = await context.newPage();
  await options.goto(`chrome-extension://${extensionId}/src/options/index.html`);
  await options.locator('.interface-language select').waitFor();
  await worker.evaluate(async origin => {
    const { translatorSettings: settings } = await chrome.storage.local.get('translatorSettings');
    await chrome.storage.local.set({ translatorSettings: { ...settings, uiLanguage: 'en', targetLanguage: 'Simplified Chinese', translationRetryCount: 0,
      profiles: [{ ...settings.profiles[0], name: 'Local fixture', provider: 'custom', protocol: 'openai', apiUrl: origin, apiKey: '', model: 'fixture' }],
    } });
    globalThis.menuUpdates = [];
    const update = chrome.contextMenus.update.bind(chrome.contextMenus);
    chrome.contextMenus.update = async (id, properties) => {
      await update(id, properties);
      globalThis.menuUpdates.push({ id, ...properties });
    };
  }, origin);
  await options.waitForFunction(() => document.documentElement.lang === 'en');
  await options.locator('input[list]').fill('unsaved-model');
  const popup = await context.newPage();
  await popup.goto(`chrome-extension://${extensionId}/src/popup/index.html`);
  await popup.locator('.popup-brand').waitFor().catch(() => popup.locator('.popup-footer').waitFor());
  for (const locale of ['zh-CN', 'en', 'fr', 'de', 'ar']) {
    await options.locator('.interface-language select').selectOption(locale);
    await options.waitForFunction(locale => document.documentElement.lang === locale, locale);
    await popup.waitForFunction(locale => document.documentElement.lang === locale, locale);
    assert.equal(await options.locator('input[list]').inputValue(), 'unsaved-model');
    const title = await options.title();
    const direction = await options.locator('html').getAttribute('dir');
    assert.equal(direction, locale === 'ar' ? 'rtl' : 'ltr');
    assert.equal(await options.locator('input[type="url"]').getAttribute('dir'), 'ltr');
    const scroll = await options.evaluate(() => ({ width: innerWidth, scroll: document.documentElement.scrollWidth }));
    assert.ok(scroll.scroll <= scroll.width, `${locale}: options horizontal overflow`);
    await options.screenshot({ path: `${output}/options-${locale}.png`, fullPage: true });
    await popup.setViewportSize({ width: 380, height: 740 });
    await popup.screenshot({ path: `${output}/popup-${locale}.png`, fullPage: true });
    const native = await worker.evaluate(async () => ({ title: await chrome.action.getTitle({}), settings: (await chrome.storage.local.get('translatorSettings')).translatorSettings }));
    assert.equal(native.settings.targetLanguage, 'Simplified Chinese');
    assert.equal(native.settings.profiles[0].model, 'fixture');
    report.locales.push({ locale, title, direction, toolbarTitle: native.title });
  }
  report.checks.push('five locales, stored preference, unchanged draft/target/model, popup live sync, LTR API URL');
  await options.setViewportSize({ width: 430, height: 900 });
  await options.emulateMedia({ colorScheme: 'dark' });
  assert.ok(await options.locator('.interface-language select').isVisible(), 'language selector must remain available on narrow screens');
  await options.screenshot({ path: `${output}/options-ar-narrow-dark.png`, fullPage: true });
  await options.setViewportSize({ width: 1280, height: 900 });
  const page = await context.newPage();
  await page.goto(origin + '/article');
  const baseline = await page.evaluate(() => ({ lang: document.documentElement.lang, dir: document.documentElement.dir, title: document.title }));
  const tabs = await worker.evaluate(origin => chrome.tabs.query({ url: origin + '/*' }), origin);
  const tabId = tabs[0].id;
  const command = (type) => worker.evaluate(({ tabId, type }) => chrome.tabs.sendMessage(tabId, { type }, { frameId: 0 }), { tabId, type });
  await page.request.get(origin + '/scenario?mode=slow');
  await command('START_TRANSLATION');
  await page.locator('[data-justranslate-state="pending"]').first().waitFor();
  const beforeSwitch = await (await page.request.get(origin + '/stats')).json();
  await options.locator('.interface-language select').selectOption('fr');
  await page.locator('[data-justranslate-state="pending"][lang="fr"]').first().waitFor();
  const afterSwitch = await (await page.request.get(origin + '/stats')).json();
  assert.equal(afterSwitch.aborted, beforeSwitch.aborted);
  await page.locator('[data-justranslate-state="translated"]').first().waitFor();
  const translatedText = await page.locator('[data-justranslate-state="translated"]').allTextContents();
  const completed = await (await page.request.get(origin + '/stats')).json();
  await options.locator('.interface-language select').selectOption('ar');
  await options.waitForFunction(() => document.documentElement.lang === 'ar');
  assert.deepEqual(await page.locator('[data-justranslate-state="translated"]').allTextContents(), translatedText);
  assert.equal(await page.locator('[data-justranslate-state="translated"]').first().getAttribute('dir'), 'auto');
  assert.equal((await (await page.request.get(origin + '/stats')).json()).calls, completed.calls);
  assert.deepEqual(await page.evaluate(() => ({ lang: document.documentElement.lang, dir: document.documentElement.dir })), { lang: baseline.lang, dir: baseline.dir });
  report.checks.push('switch during in-flight translation preserves request and completed text; host language/direction unchanged');
  await command('RESTORE_PAGE');
  await page.request.get(origin + '/scenario?mode=fail');
  await page.evaluate(() => {
    const p = document.querySelector('main p') ?? document.querySelector('p');
    const range = document.createRange(); range.selectNodeContents(p);
    const selection = getSelection(); selection.removeAllRanges(); selection.addRange(range);
    p.dispatchEvent(new MouseEvent('contextmenu', { bubbles: true, composed: true, clientX: 50, clientY: 100 }));
  });
  await worker.evaluate(({ tabId }) => chrome.tabs.sendMessage(tabId, { type: 'START_SELECTION_TRANSLATION', text: 'Selected English text for local fixture' }, { frameId: 0 }), { tabId });
  const selection = page.locator('[data-justranslate-selection]');
  await selection.locator('[data-action="retry"]').waitFor();
  await selection.locator('section').screenshot({ path: `${output}/selection-ar-error.png` });
  await options.locator('.interface-language select').selectOption('de');
  await selection.locator('section[lang="de"]').waitFor();
  assert.match(await selection.locator('.status').textContent(), /API-Anfrage fehlgeschlagen/);
  assert.equal(await selection.locator('.source').textContent(), 'Selected English text for local fixture');
  await selection.locator('section').screenshot({ path: `${output}/selection-de-error.png` });
  report.checks.push('existing selection error and buttons retranslate without replacing source or retrying');
  report.menuUpdates = await worker.evaluate(() => globalThis.menuUpdates);
  assert.ok(report.menuUpdates.some(update => update.id === 'just-translate-page' && update.title === 'Jetzt übersetzen'));
  const frame = await page.evaluateHandle(origin => {
    const iframe = document.createElement('iframe'); iframe.src = origin + '/child'; iframe.style.cssText = 'width:700px;height:400px'; document.body.append(iframe); return iframe;
  }, origin);
  const child = await (await frame.asElement().contentFrame());
  await child.waitForURL(origin + '/child');
  await child.waitForLoadState();
  // Wait for the extension's dynamically imported document-idle script to register.
  await child.waitForTimeout(500);
  await child.evaluate(() => {
    const p = document.querySelector('p'); const range = document.createRange(); range.selectNodeContents(p);
    getSelection().removeAllRanges(); getSelection().addRange(range);
    p.dispatchEvent(new MouseEvent('contextmenu', { bubbles: true, composed: true, clientX: 30, clientY: 50 }));
  });
  const frames = await worker.evaluate(tabId => chrome.webNavigation.getAllFrames({ tabId }), tabId);
  const frameId = frames.find(frame => frame.url.endsWith('/child')).frameId;
  const childResponse = await worker.evaluate(({tabId, frameId}) => chrome.tabs.sendMessage(tabId, { type: 'START_SELECTION_TRANSLATION', text: 'Iframe sample text' }, { frameId }), {tabId, frameId});
  assert.equal(childResponse.ok, true, 'child script must capture the selected range');
  await child.locator('[data-justranslate-selection] [data-action="retry"]').waitFor();
  await options.locator('.interface-language select').selectOption('fr');
  await child.locator('[data-justranslate-selection] section[lang="fr"]').waitFor();
  assert.equal(await child.locator('html').getAttribute('lang'), 'en');
  report.checks.push('existing iframe selection UI receives language broadcast without changing iframe document language');
  report.nativeMetadata = await worker.evaluate(async () => ({ name: chrome.runtime.getManifest().name, description: chrome.runtime.getManifest().description, commands: await chrome.commands.getAll() }));
  assert.ok(!JSON.stringify(report.nativeMetadata).includes('__MSG_'));
  const reopened = await context.newPage();
  await reopened.goto(`chrome-extension://${extensionId}/src/options/index.html`);
  await reopened.waitForFunction(() => document.documentElement.lang === 'fr');
  assert.equal(await reopened.locator('.interface-language select').inputValue(), 'fr');
  report.checks.push('new extension page reads durable language preference; native metadata and command descriptions resolve');
  assert.equal(errors.length, 0);
  report.checks.push('real Chrome context menu update calls succeed; no runtime page errors');
  await writeFile(`${output}/browser-results.json`, JSON.stringify(report, null, 2));
  console.log(JSON.stringify(report, null, 2));
} finally { await context.close(); }
