async (page) => {
  const origin = 'http://127.0.0.1:56634';
  const context = await page.context().browser().browserType().launchPersistentContext('/tmp/justranslate-viewport-acceptance-' + Date.now(), {
    headless: true,
    executablePath: '/Users/jintao.xu/Library/Caches/ms-playwright/chromium-1243/chrome-mac-arm64/Google Chrome for Testing.app/Contents/MacOS/Google Chrome for Testing',
    args: ['--disable-extensions-except=/Users/jintao.xu/Documents/只是翻译/dist', '--load-extension=/Users/jintao.xu/Documents/只是翻译/dist'],
    viewport: { width: 1100, height: 720 },
  });
  try {
    const worker = context.serviceWorkers()[0] || await context.waitForEvent('serviceworker');
    await worker.evaluate(async (origin) => chrome.storage.local.set({ translatorSettings: {
      profiles: [{ id: 'fixture', name: 'Local fixture', apiUrl: origin, apiKey: '', model: 'fixture', translationPrompt: 'Translate into {{targetLanguage}}.' }],
      activeProfileId: 'fixture', targetLanguage: 'Simplified Chinese', displayMode: 'bilingual', translateDynamicContent: true, excludedSites: [], autoTranslateSites: [],
    } }), origin);
    const tab = await context.newPage();
    const errors = [];
    tab.on('pageerror', error => errors.push(error.message));
    await tab.goto(origin + '/viewport');
    await tab.evaluate(() => {
      document.body.innerHTML = '<main><p id="background" style="position:absolute;top:3000px">Earlier background source.</p><p id="duplicate" style="position:absolute;top:3100px">Shared visible source.</p><p id="visible">Shared visible source.</p></main>';
    });
    const tabs = await worker.evaluate(origin => chrome.tabs.query({ url: origin + '/*' }), origin);
    const id = tabs[0].id;
    const command = type => worker.evaluate(({ id, type }) => chrome.tabs.sendMessage(id, { type }), { id, type });
    const stats = async () => (await tab.request.get(origin + '/stats')).json();
    await tab.request.get(origin + '/scenario?mode=slow');
    await command('START_TRANSLATION');
    const deadline = Date.now() + 15000;
    let first;
    let status;
    do {
      const observed = await stats();
      if (!first && observed.calls) first = observed;
      status = await command('GET_PAGE_STATUS');
      if (status.phase === 'complete' || status.phase === 'error') break;
      await tab.waitForTimeout(100);
    } while (Date.now() < deadline);
    const final = await stats();
    if (!first || first.requests[0].texts.join('') !== 'Shared visible source.') throw new Error('Initial request was not visible');
    if (status.phase !== 'complete' || status.translated !== 3 || status.failed !== 0 || final.calls !== 2 || final.aborted !== 0 || errors.length) throw new Error(JSON.stringify({status, final, errors}));
    const records = await tab.locator('p').evaluateAll(nodes => nodes.map(node => ({ id: node.id, text: node.querySelector('[data-justranslate-translation]')?.textContent })));
    if (records.some(record => !record.text)) throw new Error('A duplicate remained unrendered');
    await tab.screenshot({ path: '/Users/jintao.xu/Documents/只是翻译/artifacts/viewport-priority/review/fixed-visible-duplicate.png' });
    return { status, calls: final.calls, requests: final.requests, maximumActive: final.maximumActive, aborted: final.aborted, records, errors };
  } finally { await context.close(); }
}
