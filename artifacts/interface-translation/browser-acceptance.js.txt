async () => {
  const { TranslationController } = await import('/src/content/controller.ts');
  window.JTController = TranslationController;
  const styleModule = await import('/src/content/styles.css?inline');
  document.head.innerHTML = '<meta charset="utf-8"><title>Translation UI acceptance</title>';
  const style = document.createElement('style');
  style.textContent = styleModule.default + '\nbody{font:16px Arial;margin:32px;background:#f6f7fb;color:#222}main{background:white;padding:24px;border-radius:12px}button{height:32px;display:inline-flex;align-items:center;gap:6px;padding:0 12px;overflow:hidden;white-space:nowrap;border:1px solid #ccc;border-radius:6px;background:white}svg{width:16px;height:16px;fill:#06f}sup{background:#f33;color:white;border-radius:50%;padding:2px}form{display:flex;gap:16px;margin:20px 0}td,th{padding:12px;border-bottom:1px solid #ddd}p{margin:18px 0}nav,footer{color:#999;margin:16px 0}input{height:28px}';
  document.head.append(style);
  const html = '<nav><button>导航按钮</button></nav><main><h1>班级管理</h1><div>筛选条件：</div><form><label>班级名称<input value="保留输入" placeholder="原占位符"></label><div><span>关联课程</span></div></form><button id="create"><svg viewBox="0 0 16 16"><path d="M7 1h2v6h6v2H9v6H7V9H1V7h6z"/></svg><span>创建班级</span><sup>2</sup></button><p>这是正常的正文说明。</p><table><thead><tr><th>关联课程</th><th>操作</th></tr></thead><tbody><tr><td>三年级数学班</td><td><button id="schedule">排课</button></td></tr></tbody></table><div role="listbox" id="options" hidden><div role="option">开班状态</div></div></main><div role="dialog" id="dialog" hidden><button>确定</button></div><footer>页脚内容</footer>';
  const texts = {'班级管理':'Class Management','筛选条件：':'Filters:','班级名称':'Class Name','关联课程':'Course','创建班级':'Create class','这是正常的正文说明。':'This is the article description.','操作':'Actions','三年级数学班':'Grade 3 Math','排课':'Schedule','开班状态':'Class status','确定':'Confirm','添加学员':'Add student','异步新增':'New item'};
  const translated = (text) => texts[text] ?? ('EN ' + text);
  const settings = { configured:true,profiles:[{id:'fixture',name:'Fixture',configured:true}],activeProfileId:'fixture',targetLanguage:'English',displayMode:'bilingual',translationConcurrency:4,translationRetryCount:0,translateDynamicContent:true,autoTranslateSites:[],excludedSites:[] };
  const requests = []; let releaseFull;
  window.chrome.runtime = { sendMessage: async (request) => {
    requests.push(request);
    if (request.type === 'GET_PUBLIC_SETTINGS') return {ok:true,data:settings};
    if (request.type === 'BEGIN_TRANSLATION_SESSION') return {ok:true,data:{configurationId:'fixture-config',context:{profileId:'fixture',targetLanguage:'English'}}};
    if (request.type === 'RESOLVE_TRANSLATION_CANDIDATES') return {ok:true,data:{missIds:request.candidates.map(x=>x.id),skippedIds:[],cachedTranslations:{}}};
    if (request.type === 'TRANSLATE_BATCH') return {ok:true,data:{translations:Object.fromEntries(request.segments.map(x=>[x.requestId,translated(x.text)])),failures:{}}};
    if (request.type === 'TRANSLATE_FULL_DOCUMENT') return new Promise(resolve => {releaseFull = () => resolve({ok:true,data:Object.fromEntries(request.units.map(x=>[x.id,translated(x.text)]))});});
    return {ok:true,data:undefined};
  }};
  const results = [];
  const assert = (condition, name) => { results.push({name,passed:!!condition}); if(!condition) throw new Error(name); };
  const wait = async (predicate) => { const end=Date.now()+5000; while(!predicate()){if(Date.now()>end)throw new Error('Timed out');await new Promise(r=>setTimeout(r,30));} };
  document.body.innerHTML = html;
  let clicks = 0;
  const button = document.querySelector('#create'); button.addEventListener('click',()=>clicks++); button.focus();
  const originalButton = button.outerHTML;
  const controller = new window.JTController(); window.jtAcceptance = {controller,results,requests};
  await controller.start();
  assert(button.innerText.includes('Create class') && !button.innerText.includes('创建班级'), 'bilingual page uses translation-only button');
  assert(document.querySelector('p').innerText.includes('这是正常') && document.querySelector('p').innerText.includes('This is'), 'prose stays bilingual');
  assert(button.querySelector('svg').checkVisibility() && button.querySelector('sup').checkVisibility(), 'icons and badges visible');
  assert(button.getBoundingClientRect().height === 32 && getComputedStyle(button).flexWrap === 'nowrap', 'control geometry unchanged');
  assert(document.activeElement === button, 'focus preserved');
  button.click(); assert(clicks === 1, 'business click preserved');
  assert(document.querySelector('input').value === '保留输入' && document.querySelector('input').placeholder === '原占位符','input value and placeholder preserved');
  assert(!document.querySelector('nav [data-justranslate-translation]') && !document.querySelector('footer [data-justranslate-translation]'), 'navigation and footer excluded');
  document.querySelector('#options').hidden = false; document.querySelector('#dialog').hidden = false;
  document.querySelector('main').insertAdjacentHTML('beforeend','<button id="late">异步新增</button>');
  await wait(()=>document.querySelector('#late').innerText==='New item' && document.querySelector('#dialog').innerText==='Confirm' && document.querySelector('#options').innerText==='Class status');
  assert(true, 'hidden options, body portal and added button translated');
  button.querySelector('span:not([data-justranslate-source-content])').textContent = '添加学员';
  await wait(()=>button.innerText.includes('Add student'));
  assert(!button.innerText.includes('Create class'), 'host label replacement retranslated');
  controller.restore();
  assert(button.innerText.includes('添加学员') && !button.innerText.includes('Create class'), 'restore preserves newer host text');
  assert(!document.querySelector('[data-justranslate-translation]'), 'restore removes all output');
  document.body.innerHTML = html;
  const full = new window.JTController(); window.jtAcceptance.controller = full;
  const initialWork = full.startFullDocument(); await wait(()=>!!releaseFull);
  document.querySelector('main').insertAdjacentHTML('beforeend','<button id="full-late">异步新增</button>');
  releaseFull(); await initialWork;
  assert(full.getStatus().mode === 'full-document' && !full.getStatus().error, 'full snapshot accepts independent additions');
  await wait(()=>document.querySelector('#full-late').innerText==='New item' && full.getStatus().phase==='complete');
  assert(requests.filter(x=>x.type==='TRANSLATE_FULL_DOCUMENT').length===1, 'one full request only');
  assert(document.querySelector('#create').innerText.includes('Create class'), 'full controls are translation-only');
  full.setDisplayMode('translation'); assert(!document.querySelector('p').innerText.includes('这是正常'), 'translation-only prose toggle');
  full.setDisplayMode('bilingual');
  full.stop(); const before = requests.filter(x=>x.type==='TRANSLATE_BATCH').length;
  document.querySelector('main').insertAdjacentHTML('beforeend','<p id="stopped">停止后新增</p>');
  await new Promise(r=>setTimeout(r,1000));
  assert(requests.filter(x=>x.type==='TRANSLATE_BATCH').length===before && !document.querySelector('#stopped [data-justranslate-translation]'), 'stop disconnects dynamic work');
  const report={passed:results.length,results,status:full.getStatus(),fullRequests:requests.filter(x=>x.type==='TRANSLATE_FULL_DOCUMENT').length};
  window.jtAcceptance.report=report;
  return report;
}
