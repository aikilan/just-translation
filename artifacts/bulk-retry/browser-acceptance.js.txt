async (page) => {
  const origin = 'http://127.0.0.1:57019';
  const root = '/Users/jintao.xu/Documents/只是翻译';
  const context = await page.context().browser().browserType().launchPersistentContext('/tmp/justranslate-bulk-retry-' + Date.now(), {
    headless: true,
    executablePath: '/Users/jintao.xu/Library/Caches/ms-playwright/chromium-1243/chrome-mac-arm64/Google Chrome for Testing.app/Contents/MacOS/Google Chrome for Testing',
    args: ['--disable-extensions-except=' + root + '/dist', '--load-extension=' + root + '/dist'],
    viewport: { width: 1000, height: 760 },
  });
  try {
    const worker = context.serviceWorkers()[0] || await context.waitForEvent('serviceworker');
    const extensionId = worker.url().split('/')[2];
    await worker.evaluate(async (origin) => chrome.storage.local.set({ translatorSettings: {
      profiles: [{ id: 'fixture', name: '批量重试验收', apiUrl: origin, apiKey: '', model: 'fixture', translationPrompt: 'Translate into {{targetLanguage}}.' }],
      activeProfileId: 'fixture', targetLanguage: 'Simplified Chinese', displayMode: 'bilingual',
      translationConcurrency: 2, translationRetryCount: 0, translateDynamicContent: false,
      excludedSites: [], autoTranslateSites: [],
    } }), origin);
    const article = await context.newPage();
    const errors = [];
    article.on('pageerror', error => errors.push(error.message));
    await article.goto(origin + '/bulk');
    await article.evaluate(() => {
      document.body.innerHTML = '<main><p id="success">This paragraph has already been translated successfully.</p></main>';
    });
    const tabs = await worker.evaluate((origin) => chrome.tabs.query({ url: origin + '/*' }), origin);
    const id = tabs[0].id;
    const command = type => worker.evaluate(({ id, type }) => chrome.tabs.sendMessage(id, { type }), { id, type });
    const until = async predicate => {
      for (let i = 0; i < 180; i++) {
        const status = await command('GET_PAGE_STATUS');
        if (predicate(status)) return status;
        await article.waitForTimeout(100);
      }
      throw new Error('Status timeout: ' + JSON.stringify(await command('GET_PAGE_STATUS')));
    };
    await command('START_TRANSLATION');
    await until(status => status.phase === 'complete' && status.translated === 1);
    await article.evaluate(() => {
      window.preservedTranslation = document.querySelector('#success [data-justranslate-translation]');
      document.querySelector('main').insertAdjacentHTML('beforeend', '<p id="first">First paragraph that needs a retry.</p><p id="second">Second paragraph that needs a retry.</p>');
    });
    await article.request.get(origin + '/scenario?mode=fail');
    await command('START_TRANSLATION');
    const before = await until(status => status.phase === 'error' && status.failed === 2);
    const popup = await context.newPage();
    popup.on('pageerror', error => errors.push(error.message));
    await article.bringToFront();
    await popup.setViewportSize({ width: 380, height: 680 });
    await popup.goto('chrome-extension://' + extensionId + '/src/popup/index.html');
    await popup.getByRole('button', { name: '重试全部失败', exact: true }).waitFor();
    await popup.screenshot({ path: root + '/artifacts/bulk-retry/popup-failed-light.png' });
    await popup.emulateMedia({ colorScheme: 'dark' });
    await popup.screenshot({ path: root + '/artifacts/bulk-retry/popup-failed-dark.png' });
    await article.request.get(origin + '/scenario?mode=slow');
    await popup.getByRole('button', { name: '重试全部失败', exact: true }).click();
    await popup.getByRole('button', { name: '停止翻译', exact: true }).waitFor();
    await popup.screenshot({ path: root + '/artifacts/bulk-retry/popup-retrying.png' });
    const after = await until(status => status.phase === 'complete' && status.translated === 3);
    await popup.getByRole('button', { name: '翻译新内容', exact: true }).waitFor();
    const stats = await (await article.request.get(origin + '/stats')).json();
    const preserved = await article.evaluate(() => window.preservedTranslation === document.querySelector('#success [data-justranslate-translation]'));
    const noOverflow = await popup.evaluate(() => document.documentElement.scrollWidth <= innerWidth);
    await popup.screenshot({ path: root + '/artifacts/bulk-retry/popup-complete.png' });
    if (!preserved || !noOverflow || stats.maximumActive > 2 || stats.requests.flatMap(r => r.texts).length !== 2 || errors.length) {
      throw new Error(JSON.stringify({ preserved, noOverflow, stats, errors }));
    }
    return { before, after, preserved, noOverflow, retryTexts: stats.requests.flatMap(r => r.texts), maximumActive: stats.maximumActive, errors };
  } finally {
    await context.close();
  }
}
